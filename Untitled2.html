# ===============================
# STEP 2: Confidence Interval (Accuracy)
# ===============================

import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score

# Train model
rf = RandomForestClassifier(random_state=42)
rf.fit(X_train, y_train)

# Predictions
y_pred = rf.predict(X_test)

# Bootstrap
n_iterations = 1000
scores = []

for i in range(n_iterations):
    indices = np.random.choice(len(y_test), len(y_test), replace=True)
    
    # avoid single-class sample
    if len(np.unique(y_test.iloc[indices])) < 2:
        continue
    
    score = accuracy_score(y_test.iloc[indices], y_pred[indices])
    scores.append(score)

# Confidence interval
lower = np.percentile(scores, 2.5)
upper = np.percentile(scores, 97.5)

print("Accuracy:", accuracy_score(y_test, y_pred))
print("95% Confidence Interval: [{:.4f}, {:.4f}]".format(lower, upper))

import pandas as pd
from sklearn.metrics import confusion_matrix

def compute_metrics(y_true, y_pred):
    tn, fp, fn, tp = confusion_matrix(y_true, y_pred).ravel()
    
    accuracy = (tp + tn) / (tp + tn + fp + fn)
    sensitivity = tp / (tp + fn)  # recall
    specificity = tn / (tn + fp)
    precision = tp / (tp + fp)
    f1 = 2 * (precision * sensitivity) / (precision + sensitivity)
    mcc = ((tp * tn) - (fp * fn)) / ((tp+fp)*(tp+fn)*(tn+fp)*(tn+fn))**0.5
    bal_acc = (sensitivity + specificity) / 2
    youden = sensitivity + specificity - 1
    
    return accuracy, specificity, sensitivity, youden, precision, f1, bal_acc, mcc
